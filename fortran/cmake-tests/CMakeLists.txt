cmake_minimum_required(VERSION 3.16)
project(PlumedFortranConsumer LANGUAGES Fortran)
include(CTest)
option(TEST_CXX_LINKER "Link consumers with the kernel's C++ toolchain" OFF)
if(TEST_CXX_LINKER)
  enable_language(CXX)
endif()
if(NOT PLUMED_CONFIG)
  message(FATAL_ERROR "Set PLUMED_CONFIG to a generated or installed Plumed.cmake linking configuration")
endif()
include("${PLUMED_CONFIG}")
set(BINDING_MODULES "LEGACY;F08" CACHE STRING "Modules to exercise")
if(BINDING_MODULES STREQUAL "DEFAULT")
  plumed_add_fortran_bindings(bindings)
  set(BINDING_MODULES F08)
else()
  # Exercise duplicate removal as well as compiling both distinct modules once.
  plumed_add_fortran_bindings(bindings MODULES ${BINDING_MODULES} ${BINDING_MODULES})
endif()
add_subdirectory(consumer)
